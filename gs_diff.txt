+    fun effectiveArmorClass(character: Character): Int {

+    /** Équipe un item dans un slot. Retourne true si accepté. */
+    fun equipInSlot(characterId: String, itemName: String, slot: EquipmentSlot): Boolean {
+        val updated = _characters.value.map { character ->
+            if (character.id == characterId) {
+                val currentSlots = character.equippedSlots.toMutableMap()
+                // Règles : 1 armure max
+                if (slot == EquipmentSlot.TORSO) {
+                    val currentArmor = currentSlots[EquipmentSlot.TORSO]
+                    if (currentArmor != null && currentArmor != itemName) return@map character
+                }
+                // Arme à 2 mains nécessite main principale et main secondaire
+                if (isTwoHanded(itemName)) {
+                    if (currentSlots[EquipmentSlot.OFF_HAND] != null) return@map character
+                    currentSlots[EquipmentSlot.MAIN_HAND] = itemName
+                    currentSlots[EquipmentSlot.OFF_HAND] = itemName
+                } else {
+                    currentSlots[slot] = itemName
+                }
+                val newBackpack = character.backpackItems - itemName
+                val newEquipped = character.equippedItems - itemName + itemName
+                character.copy(
+                    backpackItems = newBackpack,
+                    equippedItems = newEquipped,
+                    equippedSlots = currentSlots
+                )
+            } else character
+        }
+        _characters.value = updated
+        saveCharacters(updated)
+        return true
+    }
